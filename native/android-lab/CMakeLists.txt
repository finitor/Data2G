cmake_minimum_required(VERSION 3.24)
project(data2g_android_lab LANGUAGES CXX)
set(DATA2G_LAB_NATIVE_SUITES ON CACHE BOOL "" FORCE)
add_subdirectory(../lab lab)
add_library(data2g_lab SHARED jni.cpp)
target_link_libraries(data2g_lab PRIVATE data2g_lab_core)
set_target_properties(data2g_lab PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)

# Execute the original suites as isolated installed PIE programs. Assertions,
# watchdog exits and crashes must not terminate the viewer/profiler process.
file(GLOB _tests "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_*.cpp")
set(_programs)
foreach(_source IN LISTS _tests)
  get_filename_component(_target "${_source}" NAME_WE)
  if(TARGET ${_target})
    target_sources(${_target} PRIVATE program_entry.cpp)
    target_compile_definitions(${_target} PRIVATE main=data2g_lab_program_main DATA2G_LAB_TEST=1)
    target_link_libraries(${_target} PRIVATE data2g_lab_core)
    set_target_properties(${_target} PROPERTIES OUTPUT_NAME "liblab_${_target}" SUFFIX ".so"
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/programs")
    list(APPEND _programs ${_target})
  endif()
endforeach()
foreach(_target IN ITEMS data2g-replay data2g-bench-rx)
  target_sources(${_target} PRIVATE program_entry.cpp)
  target_compile_definitions(${_target} PRIVATE main=data2g_lab_program_main DATA2G_LAB_ARGS=1)
  target_link_libraries(${_target} PRIVATE data2g_lab_core)
  set_target_properties(${_target} PROPERTIES OUTPUT_NAME "liblab_${_target}" SUFFIX ".so"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/programs")
  list(APPEND _programs ${_target})
endforeach()
add_custom_target(lab_programs DEPENDS ${_programs})
