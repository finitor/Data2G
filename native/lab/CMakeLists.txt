# Standalone diagnostic harness. The production core and apps are unchanged.
cmake_minimum_required(VERSION 3.24)
project(data2g_lab_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DATA2G_BUILD_PYMODULE OFF CACHE BOOL "" FORCE)
option(DATA2G_LAB_NATIVE_SUITES "Build the original hardware-free native suites" OFF)
set(DATA2G_BUILD_TESTS ${DATA2G_LAB_NATIVE_SUITES} CACHE BOOL "" FORCE)
set(DATA2G_BUILD_RIG OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_ANDROID_IO OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_QTAUDIO OFF CACHE STRING "" FORCE)
set(DATA2G_BUILD_HOST OFF CACHE STRING "" FORCE)
set(DATA2G_BUILD_GUI OFF CACHE STRING "" FORCE)
add_subdirectory(.. native EXCLUDE_FROM_ALL)
if(DATA2G_LAB_NATIVE_SUITES AND NOT ANDROID)
  file(GLOB _suite_sources "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_*.cpp")
  set(_suite_targets)
  foreach(_source IN LISTS _suite_sources)
    get_filename_component(_target "${_source}" NAME_WE)
    if(TARGET ${_target})
      list(APPEND _suite_targets ${_target})
    endif()
  endforeach()
  add_custom_target(lab_native_suites ALL DEPENDS ${_suite_targets})
endif()
add_library(data2g_lab_core STATIC channel.cpp runner.cpp metrics.cpp campaign.cpp loss.cpp)
set_target_properties(data2g_lab_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(data2g_lab_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(data2g_lab_core PUBLIC data2g_core)
if(MSVC)
  target_compile_options(data2g_lab_core PRIVATE /fp:precise /W4)
else()
  target_compile_options(data2g_lab_core PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
if(WIN32)
  target_link_libraries(data2g_lab_core PRIVATE psapi)
endif()
set(DATA2G_LAB_REVISION "unknown" CACHE STRING "Source revision and worktree fingerprint")
target_compile_definitions(data2g_lab_core PUBLIC DATA2G_LAB_REVISION="${DATA2G_LAB_REVISION}")
if(NOT ANDROID)
  add_executable(data2g-lab main.cpp)
  target_link_libraries(data2g-lab PRIVATE data2g_lab_core)
  enable_testing()
  add_executable(test_lab test_lab.cpp)
  target_link_libraries(test_lab PRIVATE data2g_lab_core)
  add_test(NAME lab COMMAND test_lab)
  set_tests_properties(lab PROPERTIES TIMEOUT 300)
  add_executable(test_lab_loss test_lab_loss.cpp)
  target_link_libraries(test_lab_loss PRIVATE data2g_lab_core)
  add_test(NAME lab_loss COMMAND test_lab_loss)
  set_tests_properties(lab_loss PROPERTIES TIMEOUT 300)
endif()
