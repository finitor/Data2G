# Standalone diagnostic harness. The production core and apps are unchanged.
cmake_minimum_required(VERSION 3.24)
project(data2g_loopback LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DATA2G_BUILD_PYMODULE OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_RIG OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_ANDROID_IO OFF CACHE BOOL "" FORCE)
set(DATA2G_BUILD_QTAUDIO OFF CACHE STRING "" FORCE)
set(DATA2G_BUILD_HOST OFF CACHE STRING "" FORCE)
set(DATA2G_BUILD_GUI OFF CACHE STRING "" FORCE)
add_subdirectory(.. native EXCLUDE_FROM_ALL)
add_library(data2g_loopback STATIC channel.cpp runner.cpp metrics.cpp)
set_target_properties(data2g_loopback PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(data2g_loopback PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(data2g_loopback PUBLIC data2g_core)
if(MSVC)
  target_compile_options(data2g_loopback PRIVATE /fp:precise /W4)
else()
  target_compile_options(data2g_loopback PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
if(WIN32)
  target_link_libraries(data2g_loopback PRIVATE psapi)
endif()
set(DATA2G_LAB_REVISION "unknown" CACHE STRING "Source revision and worktree fingerprint")
target_compile_definitions(data2g_loopback PUBLIC DATA2G_LAB_REVISION="${DATA2G_LAB_REVISION}")
if(NOT ANDROID)
  add_executable(data2g-loopback main.cpp)
  target_link_libraries(data2g-loopback PRIVATE data2g_loopback)
  enable_testing()
  add_executable(test_loopback test_loopback.cpp)
  target_link_libraries(test_loopback PRIVATE data2g_loopback)
  add_test(NAME loopback COMMAND test_loopback)
  set_tests_properties(loopback PROPERTIES TIMEOUT 300)
endif()
